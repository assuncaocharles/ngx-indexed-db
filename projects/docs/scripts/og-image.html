ngx-indexed-db

IndexedDB for Angular, without the plumbing.

Typed, Observable-based, with migrations and SSR support.

people.service.ts
db = inject(NgxIndexedDBService);

add(person: Person) {
  return this.db.add('people', person);
}

byEmail(email: string) {
  return this.db
    .getByIndex('people',
      'email', email);
}
